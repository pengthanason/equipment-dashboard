สร้าง Google Form และเชื่อม Google Sheet

สร้าง Google Form ที่มีฟิลด์: ชื่อ, นามสกุล, อุปกรณ์ที่ต้องการยืม, วันยืม, วันคืน และหมายเหตุ จากนั้นกดเปิดใช้ Google Sheets เพื่อเก็บคำตอบของแบบฟอร์ม

2

เพิ่ม Google Apps Script

ในหน้า Google Sheets คำตอบ ให้เลือกเมนู ส่วนขยาย (Extensions) > Apps Script และนำโค้ดด้านล่างนี้ไปวาง

function onSubmit(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var row = sheet.getLastRow();
  
  // ดึงค่าแถวล่าสุดที่เพิ่งกรอกเข้ามา
  var timestamp = sheet.getRange(row, 1).getValue();
  var name      = sheet.getRange(row, 2).getValue();
  var surname   = sheet.getRange(row, 3).getValue();
  var equipment = sheet.getRange(row, 4).getValue();
  var bDate     = sheet.getRange(row, 5).getValue();
  var rDate     = sheet.getRange(row, 6).getValue();
  var notes     = sheet.getRange(row, 7).getValue();

  // จัดรูปแบบวันที่
  var formattedTimestamp = Utilities.formatDate(new Date(timestamp), "GMT+7", "yyyy-MM-dd'T'HH:mm:ss");
  var formattedBDate = Utilities.formatDate(new Date(bDate), "GMT+7", "yyyy-MM-dd");
  var formattedRDate = Utilities.formatDate(new Date(rDate), "GMT+7", "yyyy-MM-dd");

  var payload = {
    "timestamp": formattedTimestamp,
    "name": name,
    "surname": surname,
    "equipment": equipment,
    "borrowDate": formattedBDate,
    "returnDate": formattedRDate,
    "notes": notes,
    "status": "กำลังยืม"
  };

  // URL ของ API ปลายทางที่ต้องการส่งไปอัปเดต (Webhook)
  var webhookUrl = "YOUR_WEBHOOK_URL_HERE";
  
  var options = {
    "method": "post",
    "contentType": "application/json",
    "payload": JSON.stringify(payload)
  };
  
  try {
    UrlFetchApp.fetch(webhookUrl, options);
  } catch(err) {
    Logger.log("Error sending: " + err.toString());
  }
}
3

ตั้งค่าทริกเกอร์ (Triggers)

ในหน้า Google Apps Script ให้ไปที่เมนู ทริกเกอร์ (Triggers) รูปนาฬิกาด้านซ้าย จากนั้นเพิ่มทริกเกอร์ใหม่ โดยตั้งค่าฟังก์ชันที่จะทำงานคือ onSubmit แหล่งข้อมูลคือ จากสเปรดชีต (From spreadsheet) และประเภทเหตุการณ์คือ เมื่อส่งฟอร์ม (On form submit)
